const loadingBox = document.getElementById('loadingBox'); const resultBox = document.getElementById('resultBox'); if(!urlInput) { alert('कृपया पहले एक टेराबॉक्स लिंक पेस्ट करें!'); return; } loadingBox.classList.remove('hidden'); resultBox.classList.add('hidden'); try { // नोट: हम बिना किसी CORS समस्या के सीधे अपने सर्वर एंडपॉइंट को हिट कर रहे हैं const response = await fetch('/api/v1/fetch', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ url: urlInput }) }); if (!response.ok) throw new Error('API सर्वर से कनेक्शन फेल हो गया।'); const result = await response.json(); loadingBox.classList.add('hidden'); if(result.status === 'success') { document.getElementById('fileName').innerText = result.name; document.getElementById('playLink').href = result.download_url; document.getElementById('downloadLink').href = result.download_url; resultBox.classList.remove('hidden'); } else { alert('एरर: लिंक प्रोसेस नहीं हो पाया। कृपया लिंक जांचें।'); } } catch (error) { loadingBox.classList.add('hidden'); alert('त्रुटि: ' + error.message); } }
